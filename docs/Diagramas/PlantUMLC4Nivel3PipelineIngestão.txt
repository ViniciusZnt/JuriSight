@startuml
skinparam linetype ortho
skinparam nodesep 150
skinparam ranksep 100
skinparam ArrowFontSize 10

!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Component.puml

LAYOUT_TOP_DOWN()

title JuriSight — Nível 3: Pipeline de Ingestão

' ── Externos ────────────────────────────────────────
System_Ext(falcao,   "Portal TST / Falcão",  "jurisprudencia.tst.jus.br")
Container(ollama,    "Ollama",               "Runtime local",    "nomic-embed-text-v2")
ContainerDb(postgres,"PostgreSQL",           "Banco Relacional", "Source of truth — DocumentoJuridico")
ContainerDb(chroma,  "ChromaDB",             "Banco Vetorial",   "Chunks + embeddings + metadados")
ContainerDb(bm25f,   "BM25 Index",           "JSON / pickle",    "Índice lexical")

Container_Boundary(pipeline, "Pipeline de Ingestão [Scrapy + Playwright]") {

    ' Fase 1 — Coleta
    Component(orch,    "Orchestrator",    "Scrapy Engine",       "Gerencia execução e controla o delta.")
    Component(delta,   "Delta Control",   "Python",              "Filtra IDs já processados\npara evitar reprocessamento.")
    Component(scraper, "Web Scraper",     "Scrapy + Playwright", "Coleta HTML e metadados do TST.\nPlaywright para páginas com JS.")

    ' Fase 2 — Extração e persistência
    Component(parser,  "HTML Extractor",  "BeautifulSoup",       "Extrai ementa, acórdão,\ntipo e hierarquia_categoria.")
    Component(clf,     "Provimento Clf",  "Regex",               "Classifica APROVADO / NEGADO /\nNAO_APLICAVEL pelo dispositivo.")
    Component(builder, "Doc Builder",     "Pydantic v2",         "Monta e valida DocumentoJuridico.")
    Component(mwriter, "DB Writer",       "psycopg2 / asyncpg",  "Persiste no PostgreSQL\ne registra o documento_id.")

    ' Fase 3 — Indexação
    Component(chunker, "SAC Chunker",     "Python",              "Divide acórdão em chunks 600–800 tokens.\nEmenta como SAC summary (prefixo).")
    Component(embedder,"Poly Embedder",   "nomic-embed-text-v2", "Gera 2 embeddings por chunk:\nembedding_completo + embedding_ementa.")
    Component(cindexer,"Vector Indexer",  "ChromaDB client",     "Indexa chunks + Poly-Vector\n+ metadados de filtragem.")
    Component(bbuilder,"BM25 Builder",    "rank_bm25",           "Constrói e persiste\nínidice lexical em JSON/pickle.")
}

' ── Layout ──────────────────────────────────────────
' Fase 1 (linha 1)
Lay_R(orch,     delta)
Lay_R(delta,    scraper)
Lay_R(scraper,  falcao)

' Fase 2 (linha 2)
Lay_R(parser,   clf)
Lay_R(clf,      builder)
Lay_R(builder,  mwriter)
Lay_R(mwriter,  postgres)

' Fase 3 (linha 3)
Lay_R(chunker,  embedder)
Lay_R(embedder, cindexer)
Lay_R(cindexer, bbuilder)

' Verticais entre fases
Lay_D(orch,     parser)
Lay_D(parser,   chunker)

' Externos das fases 3
Lay_R(bbuilder, bm25f)
Lay_D(cindexer, chroma)
Lay_D(embedder, ollama)

' ── Relações ────────────────────────────────────────
' Fase 1
Rel(orch,     delta,    "inicia")
Rel(delta,    postgres, "consulta IDs")
Rel(delta,    scraper,  "novos docs")
Rel(scraper,  falcao,   "HTTP + JS")
Rel(scraper,  parser,   "HTML bruto")

' Fase 2
Rel(parser,   clf,      "texto do dispositivo")
Rel(parser,   builder,  "ementa + acórdão")
Rel(clf,      builder,  "provimento")
Rel(builder,  mwriter,  "DocumentoJuridico")
Rel(mwriter,  postgres, "persiste")

' Fase 2 → Fase 3
Rel(mwriter,  chunker,  "id + ementa + acórdão")

' Fase 3
Rel(chunker,  embedder, "chunks + SAC")
Rel(embedder, ollama,   "gera embeddings")
Rel(embedder, cindexer, "chunks + embeddings")
Rel(embedder, bbuilder, "textos dos chunks")
Rel(cindexer, chroma,   "indexa")
Rel(bbuilder, bm25f,    "persiste")

@enduml
