title JuriSight — Nível 2: Containers

Person(user, "Advogado / Assistente Jurídico", "Usuário do Sistema")
System_Ext(falcao, "Portal Falcão / TST", "jurisprudencia.tst.jus.br")

System_Boundary(jurisight, "JuriSight") {

    Container(frontend, "Front-end", "Next.js 14", "Interface web: upload PDF, revisao de entidades, filtros e cards de resultado.")
    Container(backend, "Backend API", "FastAPI", "Query Enrichment, busca hibrida (BM25 + vetorial), RRF e ordenacao categorica.")
    Container(ollama, "Ollama Runtime", "Ollama", "Llama 3.2 3B (Query Enrichment) + nomic-embed-text-v2 (embeddings).")

    ContainerDb(postgres, "Document Database", "Postgres", "DocumentoJuridico completo (source of truth).")
    ContainerDb(chroma, "Vector Database", "ChromaDB", "Chunks + 2 embeddings/chunk + metadados de filtragem.")
    ContainerDb(bm25, "BM25 Index", "JSON/pickle", "Indice lexical (rank_bm25). Gerado pelo Pipeline, carregado pelo Backend.")
    Container(pipeline, "Pipeline de Ingestao", "Scrapy + Playwright", "Coleta, processa e indexa jurisprudencias. SAC Chunking + Poly-Vector Embeddings.")
}

' ========== RELACOES ==========
Rel_R(user, frontend, "Acessa via browser", "HTTPS")
Rel_R(frontend, backend, "REST/JSON")
Rel_R(backend, ollama, "Query Enrichment + embedding da consulta")

Rel_D(backend, postgres, "Consulta documentos por ID (pos-ranking)")
Rel_D(backend, chroma, "Busca vetorial + filtros")
Rel_D(backend, bm25, "Busca lexical (indice em memoria)")

Rel_U(pipeline, ollama, "Gera embeddings dos chunks")
Rel_U(pipeline, postgres , "Persiste DocumentoJuridico")
Rel_U(pipeline, chroma, "Indexa chunks + embeddings")
Rel_U(pipeline, bm25, "Constroi / atualiza indice BM25")

Rel_D(pipeline, falcao, "Scraping via Playwright")

SHOW_LEGEND()
@enduml
